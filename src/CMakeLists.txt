# --- Pure C++ core (no JUCE): unit-testable, strict warnings ---------------------------------
add_library(takeshape_core INTERFACE)
target_include_directories(takeshape_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(takeshape_core INTERFACE cxx_std_20)

# --- The plugin ------------------------------------------------------------------------------
juce_add_plugin(Takeshape
    PRODUCT_NAME            "Takeshape"
    COMPANY_NAME            "Takeshape"
    BUNDLE_ID               "io.github.paul-ollivier.takeshape"
    VERSION                 "${PROJECT_VERSION}"
    PLUGIN_MANUFACTURER_CODE Tksh        # 4 chars, >= 1 upper-case (all-lowercase codes are reserved by Apple)
    PLUGIN_CODE             Tksp         # 4 chars, exactly 1 upper-case (AU rule)
    FORMATS                 AU VST3 Standalone
    IS_SYNTH                FALSE
    NEEDS_MIDI_INPUT        FALSE
    NEEDS_MIDI_OUTPUT       FALSE
    IS_MIDI_EFFECT          FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE     # we'll type prompts into the editor
    AU_MAIN_TYPE            kAudioUnitType_Effect   # Logic only offers ARA on effects (aufx)
    VST3_CATEGORIES         Fx
    COPY_PLUGIN_AFTER_BUILD ${TAKESHAPE_COPY_AFTER_BUILD}

    # --- ARA ---
    IS_ARA_EFFECT           TRUE
    # JUCE gotcha: these two are pasted verbatim into a #define, so the quotes must be escaped.
    ARA_FACTORY_ID          "\"io.github.paul-ollivier.takeshape.arafactory.${PROJECT_VERSION}\""
    # Bump the trailing number whenever the archive format stops being backwards compatible.
    ARA_DOCUMENT_ARCHIVE_ID "\"io.github.paul-ollivier.takeshape.aradocumentarchive.1\""
    ARA_TRANSFORMATION_FLAGS kARAPlaybackTransformationNoChanges)

target_sources(Takeshape PRIVATE
    plugin/PluginProcessor.cpp
    plugin/PluginEditor.cpp
    ara/DocumentController.cpp
    ara/PlaybackRenderer.cpp
    ui/RegionInspector.cpp)

target_compile_definitions(Takeshape PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_STRICT_REFCOUNTEDPOINTER=1)

target_link_libraries(Takeshape
    PRIVATE
        takeshape_core
        juce::juce_audio_utils
        juce::juce_audio_formats
        juce::juce_audio_processors
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags)

# --- Xcode schemes: launch a host as the executable when you hit Run -------------------------
# Schemes are generated by CMake (never edit the .xcodeproj). Change the host with
#   cmake --preset xcode -DTAKESHAPE_DEBUG_HOST="/Applications/REAPER.app"
set(TAKESHAPE_DEBUG_HOST "/Applications/Logic Pro.app" CACHE FILEPATH "Host app launched by the AU/VST3 Xcode schemes")
foreach(fmt AU VST3)
    if(TARGET Takeshape_${fmt})
        set_target_properties(Takeshape_${fmt} PROPERTIES
            XCODE_GENERATE_SCHEME ON
            XCODE_SCHEME_EXECUTABLE "${TAKESHAPE_DEBUG_HOST}"
            XCODE_SCHEME_DEBUG_AS_ROOT OFF)
    endif()
endforeach()
if(TARGET Takeshape_Standalone)
    set_target_properties(Takeshape_Standalone PROPERTIES XCODE_GENERATE_SCHEME ON)
endif()
